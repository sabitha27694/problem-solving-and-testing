class Solution {
    public int maxSubarraySumCircular(int[] nums) {

        int n = nums.length;
        int maxSum = Integer.MIN_VALUE;

        for (int start = 0; start < n; start++) {

            int sum = 0;

            for (int len = 0; len < n; len++) {

                int index = (start + len) % n;

                sum += nums[index];

                maxSum = Math.max(maxSum, sum);
            }
        }

        return maxSum;
    }
}
